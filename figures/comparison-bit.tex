\begin{tikzpicture}[

    dependency/.style={-{Stealth[length=1.65mm,width=1.15mm]},

  }
  \foreach \k in {1,2,6,7} {
    \draw[black!25,dashed,line width=0.35pt]
      (0.6*\k,-0.3) -- (0.6*\k,0.3);
  }
  \foreach \k in {1,...,8} {
    \draw[black!30,line width=0.35pt]
      (0.6*\k,-1.25) -- (0.6*\k,-0.65);
  }
  \draw[lecol!85,line width=1pt] (4.8,-0.3) rectangle (5.4,0.3);
  \draw[gecol!85,line width=1pt] (4.8,-1.25) rectangle (5.4,-0.65);
