\section{The proof}
\label{sec:proof}
Our proof proceeds by induction on $n$, the number of
coordinates of $X,Y$. Given the random bit $C$ and the
bit string $Y\!$, we adaptively pick a coordinate $i\in[n]$ and
establish the bound
\begin{align}
  \tag{($\star$)}\label{eq:star}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid Y_i}-\muti{C}{X\emid Y_i}}
\end{align}
for some $\lambda\ge 0$.
When $Y_i$ is fixed, $C$ depends on $n-1$ coordinates and
therefore the induction hypothesis applies for $Y_i=0$ and
$Y_i=1$ separately, showing that
$\qent{C\emid Y_i}\ge \muti{C}{X\emid Y_i}$.
Plugging into \eqref{eq:star}, we get
$\qent{C}\ge\muti{C}{X}$, which implies \eqref{eq:ck}
as we show next.
\begin{lemma}
\label{lem:ck+to_ck}
For all $p\in[0,1]$, we have
$\qent{C}\le \muti{Y_1}{X}$.
\end{lemma}
\begin{proof}
From \hyperref[item:tsallis-noise]{Lemma~\ref*{lem:tsallis-properties}(\ref*{item:tsallis-noise})}
and \hyperref[item:tsallis-bounds]{(\ref*{item:tsallis-bounds})},
we have $\qJ\le\sI$ and
$\ent{C}-\tsent{C}\le1-\ent{C}$. Let us write
\begin{align*}
  \qent{C} - \sI\ent{C}
  &= \qJ\ent{C} - \qJ\tsent{C}\\
  &=   \qJ\nparen{\ent{C}-\tsent{C}}\\
  &\le \sI \nparen{1-\ent{C}}.
\end{align*}
Therefore
$\Q\le \sI\nparen{1-\ent{C}} + \sI\,\ent{C} = \muti{Y_1}{X}$
as desired.
\end{proof}
The property we want from the coordinate $i$ is fairly mild:
$\tsmuti{C}{Y_i}\le 1/4$. The next lemma shows that such a
coordinate always exists.

\begin{lemma}
\label{lem:good-coordinate}
When $n\ge2$, there exists an $i\in[n]$ such that
$\tsmuti{C}{Y_i}\le 1/4$.
\end{lemma}
\begin{proof}
Let $i,j\in[n]$ be distinct. We may relabel $Y_i,Y_j$ so that
$\E[C(2Y_i-1)]$ and $\E[C(2Y_j-1)]$ are nonnegative. Since
$Y_i,Y_j$ are independent uniform bits,
\begin{equation*}
  \sqrt{\tsmuti{C}{Y_i}}+\sqrt{\tsmuti{C}{Y_j}}
  =2\E\sparen{C(2Y_i-1)+C(2Y_j-1)}
  \le4\Pr[Y_i=Y_j=1]=1,
\end{equation*}
using $0\le C\le1$. At least one of the two square roots
is at most $1/2$.
\end{proof}
Let us fix such an $i$ satisfying $\tsmuti{C}{Y_i}\le1/4$
and use the shorthands $A=X_i$, $B=Y_i$, and $U= X_{-i}$
in the rest of the paper.
The following lemma is the crux of our argument,
which establishes \eqref{eq:star} giving the induction step.
\begin{lemma}
\label{lem:ind-step}
Assuming $\tsmuti{C}{B}\le1/4$, we have
\begin{align*}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid B}-\muti{C}{X\emid B}}
\end{align*}
for some $\lambda\ge 0$.
\end{lemma}
\autoref{sec:ind-step} is devoted to proving \autoref{lem:ind-step};
here we show that it implies the main theorem.
\begingroup
\newtheorem*{restatedtheorem}{Theorem~\ref{thm:ck+}}
\begin{restatedtheorem}[restated]
It holds that $\muti{C}{X}\le \qent{C}\le\muti{Y_1}{X}$.
\end{restatedtheorem}
\endgroup
\begin{proof}
The second inequality is \autoref{lem:ck+to_ck}.
For the first, the cases $p\in\{0,1/2,1\}$ and constant $C$
are immediate. Complementing $Y$ if necessary lets us assume
$0<p<1/2$. We proceed by induction on $n$.
When $n=1$, the bit $C$ is a constant, $Y_1$, or its negation,
and the bound holds with equality.
For $n\ge2$, by \autoref{lem:good-coordinate} there exists a coordinate
$i$ with $\tsmuti{C}{Y_i}\le1/4$.
Conditioning on $B=Y_i$ reduces the problem to $n-1$ coordinates,
so the induction hypothesis applies to the $B=0$ and $B=1$ cases
individually, giving
\begin{equation*}
  \muti{C}{X\emid B}=\muti{C}{U\emid B}
  \le\qent{C\emid B}.
\end{equation*}
Applying \autoref{lem:ind-step}, we conclude
$\qent{C}-\muti{C}{X}\ge0$.
\end{proof}

\subsection{A comparison lemma}
\label{sec:comparison}

Let $U$ be a random variable and $A,B$ be unbiased bits jointly
independent of $U$. Let $C$ be a bit obtained from $(B,U)$ using
private randomness, with $\tsent{C\emid U}>0$.
We compare $C$ with an unbiased bit $C'$ obtained from $B$ alone
using private randomness, subject to the matching condition
\begin{equation}
  \tag{(B)}
  \label{eq:comparison-match}
  \muti{B}{C'}=
  \frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}.
\end{equation}

\begin{lemma}
\label{lem:comparison}
A bit $C'$ satisfying \eqref{eq:comparison-match} exists.
Every such bit satisfies
\begin{equation*}
  \muti{A}{C'}\ge
  \frac{\muti{A}{C\emid U}}{\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
For the fixed channel between $A$ and $B$, let $F$ be the
function determined by
\begin{equation*}
F\nparen{\muti{B}{D}}=\muti{A}{D}
\end{equation*}
as $D$ ranges over unbiased bits generated from $B$.
It can be shown that $F$ is increasing and
concave (cf.\ \autoref{lem:comparison-calculus}), and
\begin{align*}
  \muti{B}{C\emid U=u}&\le\tsent{C\emid U=u},\\
  \muti{A}{C\emid U=u}
  &\le\tsent{C\emid U=u}
  F\nparen{\frac{\muti{B}{C\emid U=u}}
  {\tsent{C\emid U=u}}}.
\end{align*}
Values of $u$ with zero conditional Tsallis entropy contribute
zero to both informations and may be omitted.
The first inequality, averaged over $U$, places the right-hand
side of \eqref{eq:comparison-match} in $[0,1]$.
Varying the noise in the channel from $B$ to $C'$ therefore
gives a matching bit.
Applying Jensen's inequality to the second inequality, with weights
proportional to $\tsent{C\emid U=u}$, gives
\begin{equation*}
  \muti{A}{C\emid U}
  \le\tsent{C\emid U}
  F\nparen{\frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}}
  =\tsent{C\emid U}\muti{A}{C'},
\end{equation*}
as required.
\end{proof}

\begin{lemma}
\label{lem:comparison-correlation}
Every bit $C'$ satisfying \eqref{eq:comparison-match} also satisfies
\begin{equation*}
  \tsmuti{C'}{B}\ge
  \frac{\tsmuti{C}{B}^{\phantom{2}}}{\tsent{C\emid U}^{2}}.
\end{equation*}
\end{lemma}
\begin{proof}
Let us write $v=\tsent{C\emid U}$ and
$d=\E[(2B-1)(2C-1)]$.
Since $B$ is unbiased, $\tsmuti{C}{B}=d^2$.
As shown in \autoref{app:comparison}, $|d|\le v$ and
\begin{equation*}
  \frac{\muti{B}{C\emid U}}v
  \ge\kldivb{(1+d/v)/2}{1/2}.
\end{equation*}
If $\theta=\E[(2B-1)(2C'-1)]$, then
$\tsmuti{C'}{B}=\theta^2$ and
$\muti{B}{C'}=\kldivb{(1+\theta)/2}{1/2}$.
The matching identity and monotonicity of this divergence in
$|\theta|$ imply $\theta^2\ge d^2/v^2$.
\end{proof}

\subsection{The induction step}
\label{sec:ind-step}
Recalling that $A=X_i$, $B=Y_i$ and $U=X_{-i}$,
in this section we would like to lower bound the gap
$\qent{C} - \muti{C}{X}$ by a factor of the conditional gap
\begin{align*}
\Delta=\qent{C\emid B} - \muti{C}{X\emid B}.
\end{align*}

We obtain two lower bounds on $\qent{C}-\muti{C}{X}$.
The first involves $\Delta$ and benefits from $C$ depending on
$U$, through a positive term in $\tsmuti{C}{U}$.
The second has a negative term in $\tsmuti{C}{U}$ and
benefits from weaker dependence.
We add $\lambda$ times the
second bound to the first, choosing $\lambda$ to cancel these
terms, so that the bound works whether or not $C$ depends on $U$.

To analyze the remaining terms, we introduce an unbiased
comparison bit $C'$ obtained from $B$ using private randomness.
The comparisons in \autoref{lem:comparison} and
\autoref{lem:comparison-correlation}, together with our choice
of coordinate with $\tsmuti{C}{B}\le1/4$, allow us to charge
these terms to quantities
involving $\muti{A}{C'}$ and $\muti{B}{C'}$.
What remains is an inequality about three unbiased bits 
forming the Markov chain $A-B-C'$, which we analyze in
\autoref{sec:3bit}.

\input{figures/comparison-bit}

Let us start with a lemma that we use in obtaining the first
lower bound.
\begin{lemma}
\label{lem:conditioning-gain}
If $\tsent{C\emid U}>0$, then
\begin{equation*}
  \muti{B}{C\emid U}-\muti{B}{C}
  \ge
  \frac{1}{2\ln 2}
  \frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
  {\tsent{C}\,\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
Since $B$ and $U$ are independent, the Shannon and Tsallis
chain rules give
\begin{align*}
  \muti{B}{C\emid U}-\muti{B}{C}
  &=\muti{B}{U\emid C},\\
  \tsmuti{B}{C\emid U}-\tsmuti{B}{C}
  &=\tsmuti{B}{U\emid C}.
\end{align*}
Since $B$ is unbiased and independent of $U$, parts
\ref{item:tsallis-shannon}--\ref{item:tsallis-conditioning}
of \autoref{lem:tsallis-properties} give
\begin{align*}
  \muti{B}{U\emid C}
  &\ge\frac{\tsmuti{B}{U\emid C}}{2\ln2}\\
  &\ge\frac{\tsmuti{C}{B}}{2\ln2}
  \sparen{\frac1{\tsent{C\emid U}}-\frac1{\tsent{C}}}\\
  &=\frac{1}{2\ln 2}
    \frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
         {\tsent{C}\,\tsent{C\emid U}},
\end{align*}
as required.
\end{proof}

When $\tsent{C\emid U}>0$, let $C'$ be an unbiased bit obtained
from $B$ alone using private randomness and satisfying
\eqref{eq:comparison-match}.

\begin{lemma}
\label{lem:gap-first}
If $\tsent{C\emid U}>0$, then
\begin{equation}
  \label{eq:gap-first}
  \begin{aligned}
    \qent{C}-\muti{C}{X}\ge \Delta
    &+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}\\
    &+\frac{(1-\qI)\tsmuti{C}{B}\,\tsmuti{C}{U}}
    {2\ln2\,\tsent{C}\,\tsent{C\emid U}}
    -\qJ\tsmuti{C}{B}.
  \end{aligned}
\end{equation}
\end{lemma}
\begin{proof}
The chain rules for Shannon and Tsallis information give
\begin{align*}
  \qent{C}-\muti{C}{X}-\Delta
  =\muti{C}{B\emid X}
  -(1-\qI)\muti{B}{C}-\qJ\tsmuti{C}{B}.
\end{align*}
The chain rule and \autoref{lem:comparison} give
\begin{align*}
  \muti{C}{B\emid X}
  &=\muti{B}{C\emid U}-\muti{A}{C\emid U}\\
  &\ge(1-\qI)\muti{B}{C\emid U}
  +\tsent{C\emid U}
  \sparen{\qI\muti{B}{C'}-\muti{A}{C'}}.
\end{align*}
Substituting this into the preceding identity and applying
\autoref{lem:conditioning-gain} gives \eqref{eq:gap-first}.
\end{proof}

\begin{lemma}
\label{lem:gap-second}
If $\tsent{C\emid U}>0$, then
\begin{align}
  \label{eq:gap-second}
  \qent{C}-\muti{C}{X}
  &\ge\tsent{C\emid U}\muti{A}{B\emid C'}
  -\sparen{\frac{(1-\qI)c(\mu)}{\ln2}+\qJ}\tsmuti{C}{U}.
\end{align}
\end{lemma}
\begin{proof}
Using the comparison in
$\muti{C}{X}=\muti{C}{U}+\muti{A}{C\emid U}$ gives
\begin{align*}
  \qent{C}-\muti{C}{X}
  &\ge\qI\ent{C}-\qJ\tsent{C}
  -\muti{C}{U}-\tsent{C\emid U}\muti{A}{C'}\\
  &=\tsent{C\emid U}\muti{A}{B\emid C'}
  +\qI\sparen{\ent{C\emid U}-\tsent{C\emid U}}\\
  &\quad-(1-\qI)\muti{C}{U}-\qJ\tsmuti{C}{U}.
\end{align*}
Here we used $\sI=\muti{A}{B}$ and
$\muti{A}{B\emid C'}=\muti{A}{B}-\muti{A}{C'}$.
Now $\ent{C\emid U}\ge\tsent{C\emid U}$ and
\autoref{lem:biased-comparison} give \eqref{eq:gap-second}.
\end{proof}

For $0<p<1$, we use the shorthand
\begin{equation*}
  \alpha_p=\frac{\qJ\ln2}{1-\qI}.
\end{equation*}
We need the following fact about the Markov chain $A-B-C'$
of three unbiased bits. We state it here and prove it in
\autoref{sec:3bit}.

\begin{lemma}
\label{lem:three-bit}
Let $C'$ be an unbiased bit obtained from $B$ using private
randomness. If $0<\qI<1$ and $\tsmuti{B}{C'}>0$, then for
every $0<d\le1/2$,
\begin{align*}
  \frac{\sI-\muti{A}{C'}}
  {2(1+\alpha_p-d/4)}
  +
  \frac{\qI\muti{B}{C'}-\muti{A}{C'}}
  {d\sqrt{\tsmuti{B}{C'}}}
  \ge\qJ.
\end{align*}
\end{lemma}

\begin{proof}[Proof of \autoref{lem:ind-step}]
The cases $p\in\{0,1/2,1\}$ are immediate, so we may assume
$0<p<1/2$, complementing $Y$ if necessary.
If $\tsmuti{C}{B}=0$, then $C$ and $B$ are independent.
Consequently $\qent{C}=\qent{C\emid B}$ and
$\muti{C}{X\emid B}\ge\muti{C}{X}$, proving the claim with
$\lambda=0$.

We may therefore assume $\tsmuti{C}{B}>0$.
Then $\tsent{C\emid U}>0$:
otherwise $C$ would be determined by $U$ and hence independent of
$B$. By \autoref{lem:comparison}, there is an unbiased bit $C'$
obtained from $B$ alone and satisfying \eqref{eq:comparison-match}.
Let $d\in(0,1/2]$ satisfy $d^2=\tsmuti{C}{B}$, and set
\begin{equation*}
  \lambda=
  \frac{\tsmuti{C}{B}}
  {2\tsent{C}\,\tsent{C\emid U}
  \nparen{c(\mu)+\alpha_p}}.
\end{equation*}
All factors are positive. When we multiply \eqref{eq:gap-second} by
$\lambda$ and add it to \eqref{eq:gap-first},
the terms involving $\tsmuti{C}{U}$ cancel, giving
\begin{equation}
  \label{eq:gap-combined}
  \begin{aligned}
    (1+\lambda)\nparen{\qent{C}-\muti{C}{X}}\ge\Delta
    &+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}\\
    &+\lambda\tsent{C\emid U}\muti{A}{B\emid C'}
    -\qJ\tsmuti{C}{B}.
  \end{aligned}
\end{equation}

By \autoref{lem:comparison-correlation},
\begin{equation*}
  \frac{\tsent{C\emid U}}{\tsmuti{C}{B}}
  \ge\frac1{d\sqrt{\tsmuti{B}{C'}}}.
\end{equation*}
Since $B$ is unbiased, $d\le2\min\{\mu,1-\mu\}$.
Complementing $C$ if necessary, assume $\mu\ge1/2$, so
$\mu\le1-d/2$. Using $\ln x\le(x-x^{-1})/2$ for $x\ge1$ gives
\begin{equation*}
  \tsent{C}\,c(\mu)
  =\frac{\mu\ln(1/\mu)}{1-\mu}
  \le\frac{1+\mu}{2}\le1-\frac d4.
\end{equation*}
Consequently, using $\tsent{C}\le1$,
\begin{equation*}
  \frac{\lambda\tsent{C\emid U}}{\tsmuti{C}{B}}
  =\frac1{2\tsent{C}
  \nparen{c(\mu)+\alpha_p}}
  \ge\frac1{2(1+\alpha_p-d/4)}.
\end{equation*}
The two information quantities multiplied by these coefficients
are nonnegative. For the first, the binary divergence series gives
$\muti{A}{C'}\le\tsmuti{A}{B}\muti{B}{C'}$; the second
is a conditional mutual information.
Applying \autoref{lem:three-bit}, we therefore obtain
\begin{align*}
  &\frac{\tsent{C\emid U}}{\tsmuti{C}{B}}
  \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}
  +\frac{\lambda\tsent{C\emid U}}{\tsmuti{C}{B}}
  \muti{A}{B\emid C'}\ge\qJ.
\end{align*}
Thus the last three terms in \eqref{eq:gap-combined} have
nonnegative sum, so
$\qent{C}-\muti{C}{X}\ge\Delta/(1+\lambda)$, as required.
\end{proof}

\subsection{The three-bit lemma}
\label{sec:3bit}
In this section we study the Markov chain $A-B-C$ of three
unbiased bits. The joint law is determined by two parameters:
the correlations $\rho$ of $A,B$ and $\sigma$ of $B,C$.
These satisfy $\rho^2=\tsmuti{A}{B}$ and $\sigma^2=\tsmuti{B}{C}$.
Assume $\rho^2\neq 1$, $\sigma\neq 0$ and fix a $d\in (0,1/2]$.
\input{figures/three-bit-chain}
To prove \autoref{lem:three-bit}, we expand the binary divergence
$\kldivb{x}{1/2}$ in a Taylor series about $x=1/2$. Writing
$t=2x-1$ for the bias, we define
\begin{equation*}
  f(t)=\kldivb{(1+t)/2}{1/2},\qquad
  \alpha=\frac{(\rho^2-f(\rho))\ln2}{1-\rho^2}.
\end{equation*}
Then $\muti{A}{B}=f(\rho)$, $\muti{B}{C}=f(\sigma)$,
and $\muti{A}{C}=f(\rho\sigma)$, so it suffices to prove
\begin{equation}
  \label{eq:three-bit-scalar-target}
  \frac{f(\rho)-f(\rho\sigma)}{2(1+\alpha-d/4)}
  +\frac{\rho^2 f(\sigma)-f(\rho\sigma)}{d\abs{\sigma}}
  \ge\rho^2-f(\rho).
\end{equation}
When $\rho=0$, both sides vanish. Since $f$ is even, it suffices
to prove the strict inequality for $\rho,\sigma>0$.
We expand in $\rho^2$ and prove positivity of every coefficient.
For $N\ge1$, define the divergence tails
\begin{align*}
  b_N&=\sum_{k>N}\frac1{2k(2k-1)}
      =\int_0^1\frac{x^{2N}}{1+x}\,\mathrm dx,\\
  s_N(\sigma)&=\sum_{k>N}\frac{\sigma^{2k}}{2k(2k-1)}
      =\int_0^\sigma\frac{(\sigma-x)x^{2N}}{1-x^2}\,\mathrm dx.
\end{align*}
Set $b_0=1-d/4$ as a separate convention, and write
\begin{align*}
  B_N&=\sum_{i=0}^{N-1}b_i b_{N-i},
  &G_N(\sigma)&=\sum_{i=0}^{N-1}b_i s_{N-i}(\sigma),\\
  e_N&=\frac12\sum_{k=1}^N\frac1{2k-1},
  &C_N(\sigma)&=\sum_{k=1}^N\frac{1-\sigma^{2k}}{2k(2k-1)}.
\end{align*}
The first ingredient compares the divergence tails in two ways.
\begin{lemma}
\label{lem:scalar-tail-order}
For $0<\sigma<1$ and $N\ge1$,
\begin{equation}
  \label{eq:scalar-convolution-order}
  G_N(\sigma)\ge B_N\frac{s_N(\sigma)}{b_N}.
\end{equation}
For $N\ge2$, we also have
\begin{equation}
  \label{eq:scalar-rescaled-order}
  \frac{s_N(\sigma)}{\sigma b_N}\ge\frac{s_2(\sigma^{N/2})}{\sigma^{N/2} b_2}.
\end{equation}
\end{lemma}
\begin{proof}
First, $s_j(\sigma)/b_j$ is the expectation of the decreasing function
$(\sigma-x)_+/(1-x)$ under the probability density proportional to
$x^{2j}/(1+x)$ on $(0,1)$. Increasing $j$ tilts this density by an
increasing function, so $s_j(\sigma)/b_j$ decreases with $j$.
For $1\le j\le N$, therefore,
$s_j(\sigma)\ge b_j s_N(\sigma)/b_N$; summing proves
\eqref{eq:scalar-convolution-order}.
Here is the elementary order principle used in this argument.
If $Z,Z'$ are independent with the same distribution, then
\begin{equation*}
  \operatorname{Cov}(f(Z),g(Z))
  =\frac12\E\sparen{(f(Z)-f(Z'))(g(Z)-g(Z'))}.
\end{equation*}
It is nonnegative when both functions increase and nonpositive when
one increases and the other decreases. Applying this identity to a
density ratio gives the stated comparison of expectations.
For the second comparison, let us fix $a\ge0$ and write
\begin{equation*}
  M_N(a)=\frac{s_N(e^{-a/N})}{e^{-a/N}b_N}.
\end{equation*}
Changing variables in the two tail integrals gives
\begin{equation}
  \label{eq:scalar-tail-kernel}
  M_N(a)=e^{-2a}\E_{\pi_N}
  \sparen{\frac{\sinh(z/N)}{\sinh((z+a)/N)}},
  \qquad
  \pi_N(\mathrm{d}z)\ \propto
  \frac{e^{-2z}}{1+e^{z/N}}\,\mathrm{d}z,
  \quad z>0.
\end{equation}
The quotient inside the expectation increases with $z$ because
$\coth$ decreases. It also increases with $N$, since
$x\coth x$ increases for $x>0$:
its derivative has the sign of $\sinh x\cosh x-x>0$.
Moreover, for $K>N$ the density ratio $\pi_K/\pi_N$ increases with
$z$, as its logarithmic derivative is
\begin{equation*}
  \frac{N^{-1}}{1+e^{-z/N}}
  -\frac{K^{-1}}{1+e^{-z/K}}>0.
\end{equation*}
The order principle and then the pointwise comparison of quotients
show that $M_N(a)$ increases with $N$.
Taking $a=-N\ln \sigma$ gives
$e^{-a/N}=\sigma$ and $e^{-a/2}=\sigma^{N/2}$, proving
\eqref{eq:scalar-rescaled-order}.
All the densities are integrable; at $a=0$ the quotient in
\eqref{eq:scalar-tail-kernel} equals one.
\end{proof}

We will use the elementary bounds
\begin{equation*}
  \frac{11}{16}<\ln2<\frac{25}{36}.
\end{equation*}
Both follow from
$\ln2=2\sum_{j\ge0}[(2j+1)3^{2j+1}]^{-1}$:
keep its first two terms for the lower bound, and replace
$2j+1$ by $3$ for $j\ge1$ for the upper bound.
The next lemma bounds the two coefficient sums.
\begin{lemma}\label{lem:scalar-moments}
For $N\ge2$, $0\le \sigma\le1$, and $x=1-\sigma^{2N}$,
\begin{equation*}
 \frac{B_N}{2e_Nb_N}\le\frac{11}{10}-\frac d5,
 \qquad
 \frac{C_N(\sigma)}{2e_Nb_N}\ge2x+\frac23x^2.
\end{equation*}
\end{lemma}
\begin{proof}
The tail integral, using $1+x\le2$ and $1+x\ge2\sqrt{x}$, gives
$1/(4N+2)\le b_N\le1/(4N+1)$. These bounds imply
$b_i b_j\le b_{i+j}(b_i+b_j)$ for positive $i,j$.
Also $\sum_{i=1}^{N-1}b_i=e_N+Nb_N-\ln2$.
Thus
\begin{equation*}
 \frac{B_N}{2e_Nb_N}
 \le1+\frac{Nb_N-(\ln2-1/2)-d/8}{e_N}
 \le\frac{35}{32}-\frac{3d}{16}
 \le\frac{11}{10}-\frac d5.
\end{equation*}
For the middle inequality use $Nb_N\le1/4$, $\ln2>11/16$,
$e_N\ge2/3$, and $d\le1/2$. The resulting upper numerator
$1/16-d/8$ is nonnegative, so using $e_N\ge2/3$ preserves
the direction of the estimate.

For the second claim, use the first two terms of
$1-(1-x)^{k/N}$, the identity
$\sum_{k=1}^N k/[2(2k-1)]=N/4+e_N/2$,
and $b_N\le1/(4N+1)$. With $s=1/N$, these give
\begin{equation*}
 \frac{C_N(\sigma)}{2e_Nb_N}
 \ge\frac{4+s}{2}\sparen{x+
 \nparen{\frac12-\frac s4-\frac1{8e_N}}x^2}.
\end{equation*}
Since $x\ge x^2$, it suffices that
\begin{equation*}
 \frac s2+\frac{4+s}{2}
 \nparen{\frac12-\frac s4-\frac1{8e_N}}\ge\frac23.
\end{equation*}
At $N=2$, the left side is $43/64>2/3$.
For $N\ge3$, use $e_N\ge23/30$ and $s\le1/3$ to bound it below by
\begin{equation*}
 \frac{31}{46}+\frac{31s}{184}-\frac{s^2}{8}
 \ge\frac{31}{46}>\frac23.
\end{equation*}
\end{proof}

\begin{lemma}\label{lem:scalar-fixed-function}
For $0<w<1$, writing $x=1-w^4$,
\begin{equation*}
 2x+\frac23x^2+36\frac{s_2(w)}w-2>0.
\end{equation*}
\end{lemma}
\begin{proof}
Keeping just the terms of $s_2$ indexed by $3\le k\le7$
bounds the left side below by the rational polynomial
\begin{equation*}
 P(w)=\frac23-\frac{10}{3}w^4+\frac23w^8
       +36\sum_{k=3}^7\frac{w^{2k-1}}{2k(2k-1)}.
\end{equation*}
The ratio
\begin{equation*}
 \frac{P'(w)}{w^3}
 =-\frac{40}{3}+\frac{16}{3}w^4
   +18\sum_{k=3}^7\frac{w^{2k-5}}k
\end{equation*}
is strictly increasing. Also $P'''(w)>0$ for $w\ge4/5$:
the polynomial part contributes $16w(14w^4-5)>0$, and all
tail terms contribute positively. Direct rational arithmetic gives
\begin{equation*}
 P'(4/5)<0,\quad P''(4/5)>18,\quad
 P(5/6)>1/128,\quad \abs{P'(5/6)}<1/4.
\end{equation*}
Taylor's formula therefore gives, for $4/5\le w<1$,
\begin{equation*}
 P(w)\ge P(5/6)+P'(5/6)(w-5/6)+9(w-5/6)^2
 >\frac1{128}-\frac1{576}>0.
\end{equation*}
On $(0,4/5]$, monotonicity of $P'(w)/w^3$ makes $P$
decreasing, so $P(w)\ge P(4/5)>0$ there too.
\end{proof}

\begin{lemma}\label{lem:scalar-coefficients}
For every $N\ge1$ and $0<\sigma\le1$,
\begin{equation*}
 C_N(\sigma)+\frac{2G_N(\sigma)}{d\sigma}-2B_N>0.
\end{equation*}
\end{lemma}
\begin{proof}
First suppose $N\ge2$ and $\sigma<1$.
By \autoref{lem:scalar-tail-order}, the expression is at least
\begin{equation*}
 C_N(\sigma)+B_N\sparen{\frac{2s_N(\sigma)}{d\sigma b_N}-2}.
\end{equation*}
If the bracket is nonnegative, we are done. Otherwise apply
\autoref{lem:scalar-moments}, remembering that an upper bound on $B_N$
gives a lower bound after multiplication by this negative bracket.
Put $w=\sigma^{N/2}$ and $x=1-w^4$.
Since $b_2=\ln2-7/12<1/9$, \autoref{lem:scalar-tail-order} shows that the
expression, divided by $2e_Nb_N$, is at least
\begin{equation}\label{eq:scalar-coefficient-lower}
 2x+\frac23x^2+
 \nparen{\frac{11}{10}-\frac d5}
 \sparen{\frac{18s_2(w)}{dw}-2}.
\end{equation}
If $s_2(w)/w\ge1/198$, this expression is decreasing in
$d\in(0,1/2]$: its derivative is
\begin{equation*}
 -\frac{99}{5d^2}\frac{s_2(w)}w+\frac25\le0.
\end{equation*}
Its value is therefore at least its value at $d=1/2$, which is
positive by \autoref{lem:scalar-fixed-function}.

Suppose instead that $s_2(w)/w<1/198$. The first two tail terms
at $w=2/3$ already exceed $1/198$, so $w<2/3$.
If $w\le3/5$, then $x\ge6/7$, and dropping the positive terms
in the expansion of \eqref{eq:scalar-coefficient-lower} gives the lower bound
\begin{equation*}
 2x+\frac23x^2-\frac{11}{5}
 \ge\frac{108}{49}-\frac{11}{5}>0.
\end{equation*}
For $3/5<w<2/3$, we have $x>4/5$ and
$s_2(w)/w>w^5/30>1/400$. By AM--GM,
\begin{equation*}
 \frac{99}{5d}\frac{s_2(w)}w+\frac{2d}{5}
 \ge\sqrt{\frac{792}{25}\frac{s_2(w)}w}>\frac7{25}.
\end{equation*}
Also $-(18/5)s_2(w)/w>-1/55$.
Expanding \eqref{eq:scalar-coefficient-lower} now gives the strict lower bound
\begin{equation*}
 2-\frac{11}{5}+\frac7{25}-\frac1{55}>0.
\end{equation*}

At $\sigma=1$, we have $C_N(1)=0$ and $G_N(1)=B_N$, giving
$(2/d-2)B_N>0$ for every $N$.

For $N=1$ and $\sigma<1$, the coefficient is
\begin{equation*}
 \frac{1-\sigma^2}{2}+\nparen{1-\frac d4}
 \sparen{\frac{2s_1(\sigma)}{d\sigma}-2b_1}.
\end{equation*}
If the bracket is nonnegative this is positive. Otherwise,
using $1-d/4\le1$ and $2/d\ge4$, it is at least
$(1-\sigma^2)/2+4s_1(\sigma)/\sigma-2b_1$. Now
$b_1<7/36$ and the first two tail terms bound this below by
\begin{equation*}
 Z(\sigma)=\frac19-\frac{\sigma^2}{2}+\frac{\sigma^3}{3}+\frac{2\sigma^5}{15}>0.
\end{equation*}
Indeed, $Z'(\sigma)/\sigma=-1+\sigma+2\sigma^3/3$ increases and is negative at
$2/3$, whereas $Z''>1$ on $[2/3,1]$.
At $a=3/4$, $Z(a)=49/23040>1/512$ and
$Z'(a)=3/128<1/32$. Taylor's formula gives
$Z(\sigma)>1/512-1/2048>0$ on $[2/3,1]$, and monotonicity
gives positivity on the remaining interval.
\end{proof}

\newtheorem*{restatedthreebitlemma}{Lemma~\ref{lem:three-bit}}
\begin{restatedthreebitlemma}[restated]
Let $A-B-C$ be a Markov chain of unbiased bits with
$\tsmuti{A}{B}<1$ and $\tsmuti{B}{C}>0$. For every $0<d\le1/2$,
\begin{equation*}
  \frac{\muti{A}{B\emid C}}
  {2\nparen{1-\frac d4+
    \frac{\nparen{\tsmuti{A}{B}-\muti{A}{B}}\ln2}
    {\tsent{A\emid B}}}}
  +\frac{\tsmuti{A}{B}\muti{B}{C}-\muti{A}{C}}
  {d\sqrt{\tsmuti{B}{C}}}
  \ge\tsmuti{A}{B}-\muti{A}{B}.
\end{equation*}
\end{restatedthreebitlemma}
\endgroup
\begin{proof}
The divergence-tail definitions give the absolutely convergent expansions
\begin{align*}
 \alpha&=\sum_{N\ge1}b_N\rho^{2N},\\
 \frac{\ln2}{1-\rho^2}\sparen{\rho^2 f(\sigma)-f(\rho\sigma)}
 &=\sum_{N\ge1}s_N(\sigma)\rho^{2N},\\
 \frac{\ln2}{1-\rho^2}\sparen{f(\rho)-f(\rho\sigma)}
 &=\sum_{N\ge1}C_N(\sigma)\rho^{2N}.
\end{align*}
Multiplying \eqref{eq:three-bit-scalar-target}, after moving its right side to the left,
by $2(1-d/4+\alpha)\ln2/(1-\rho^2)$ gives
\begin{equation*}
 \sum_{N\ge1}
 \sparen{C_N(\sigma)+\frac{2G_N(\sigma)}{d\sigma}-2B_N}\rho^{2N}>0
\end{equation*}
by \autoref{lem:scalar-coefficients}.
\end{proof}
