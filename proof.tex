\section{The proof}
\label{sec:proof}
Our proof proceeds by induction on $n$, the number of
coordinates of $X,Y$. Given the random bit $C$ and the
bit string $Y\!$, we adaptively pick a coordinate $i\in[n]$ and
establish the bound
\begin{align}
  \tag{($\star$)}\label{eq:star}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid Y_i}-\muti{C}{X\emid Y_i}}
\end{align}
for some $\lambda\ge 0$.
When $Y_i$ is fixed, $C$ depends on $n-1$ coordinates and
therefore the induction hypothesis applies for $Y_i=0$ and
$Y_i=1$ separately, showing that
$\qent{C\emid Y_i}\ge \muti{C}{X\emid Y_i}$.
Plugging into \eqref{eq:star}, we get
$\qent{C}\ge\muti{C}{X}$, which implies \eqref{eq:ck}
as we show next.
\begin{lemma}
\label{lem:ck+to_ck}
For all $p\in[0,1]$, we have
$\qent{C}\le \muti{Y_1}{X}$.
\end{lemma}
\begin{proof}
From \hyperref[item:tsallis-noise]{Lemma~\ref*{lem:tsallis-properties}(\ref*{item:tsallis-noise})}
and \hyperref[item:tsallis-bounds]{(\ref*{item:tsallis-bounds})},
we have $\qJ\le\sI$ and
$\ent{C}-\tsent{C}\le1-\ent{C}$. Let us write
\begin{align*}
  \qent{C} - \sI\ent{C}
  &= \qJ\ent{C} - \qJ\tsent{C}\\
  &=   \qJ\nparen{\ent{C}-\tsent{C}}\\
  &\le \sI \nparen{1-\ent{C}}.
\end{align*}
Therefore
$\Q\le \sI\nparen{1-\ent{C}} + \sI\,\ent{C} = \muti{Y_1}{X}$
as desired.
\end{proof}
The property we want from the coordinate $i$ is fairly mild:
$\tsmuti{C}{Y_i}\le 1/4$. The next lemma shows that such a
coordinate always exists.

\begin{lemma}
\label{lem:good-coordinate}
When $n\ge2$, there exists an $i\in[n]$ such that
$\tsmuti{C}{Y_i}\le 1/4$.
\end{lemma}
\begin{proof}
Let $i,j\in[n]$ be distinct. We may relabel $Y_i,Y_j$ so that
$\E[C(2Y_i-1)]$ and $\E[C(2Y_j-1)]$ are nonnegative. Since
$Y_i,Y_j$ are independent uniform bits,
\begin{equation*}
  \sqrt{\tsmuti{C}{Y_i}}+\sqrt{\tsmuti{C}{Y_j}}
  =2\E\sparen{C(2Y_i-1)+C(2Y_j-1)}
  \le4\Pr[Y_i=Y_j=1]=1,
\end{equation*}
using $0\le C\le1$. At least one of the two square roots
is at most $1/2$.
\end{proof}
Let us fix such an $i$ satisfying $\tsmuti{C}{Y_i}\le1/4$
and use the shorthands $A=X_i$, $B=Y_i$, and $U= X_{-i}$
in the rest of the paper.
The following lemma is the crux of our argument,
which establishes \eqref{eq:star} giving the induction step.
\begin{lemma}
\label{lem:ind-step}
Assuming $\tsmuti{C}{B}\le1/4$, we have
\begin{align*}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid B}-\muti{C}{X\emid B}}
\end{align*}
for some $\lambda\ge 0$.
\end{lemma}
\autoref{sec:ind-step} is devoted to proving \autoref{lem:ind-step};
here we show that it implies the main theorem.
\begingroup
\newtheorem*{restatedtheorem}{Theorem~\ref{thm:ck+}}
\begin{restatedtheorem}[restated]
It holds that $\muti{C}{X}\le \qent{C}\le\muti{Y_1}{X}$.
\end{restatedtheorem}
\endgroup
\begin{proof}
The second inequality is \autoref{lem:ck+to_ck}.
For the first, the cases $p\in\{0,1/2,1\}$ and constant $C$
are immediate. Complementing $Y$ if necessary lets us assume
$0<p<1/2$. We proceed by induction on $n$.
When $n=1$, the bit $C$ is a constant, $Y_1$, or its negation,
and the bound holds with equality.
For $n\ge2$, by \autoref{lem:good-coordinate} there exists a coordinate
$i$ with $\tsmuti{C}{Y_i}\le1/4$.
Conditioning on $B=Y_i$ reduces the problem to $n-1$ coordinates,
so the induction hypothesis applies to the $B=0$ and $B=1$ cases
individually, giving
\begin{equation*}
  \muti{C}{X\emid B}=\muti{C}{U\emid B}
  \le\qent{C\emid B}.
\end{equation*}
Applying \autoref{lem:ind-step}, we conclude
$\qent{C}-\muti{C}{X}\ge0$.
\end{proof}

\subsection{A comparison lemma}
\label{sec:comparison}

Let $U$ be a random variable and $A,B$ be unbiased bits jointly
independent of $U$. Let $C$ be a bit obtained from $(B,U)$ using
private randomness, with $\tsent{C\emid U}>0$.
We compare $C$ with an unbiased bit $C'$ obtained from $B$ alone
using private randomness, subject to the matching condition
\begin{equation}
  \tag{(B)}
  \label{eq:comparison-match}
  \muti{B}{C'}=
  \frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}.
\end{equation}

\begin{lemma}
\label{lem:comparison}
A bit $C'$ satisfying \eqref{eq:comparison-match} exists.
Every such bit satisfies
\begin{equation*}
  \muti{A}{C'}\ge
  \frac{\muti{A}{C\emid U}}{\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
For the fixed channel between $A$ and $B$, let $F$ be the
function determined by
\begin{equation*}
F\nparen{\muti{B}{D}}=\muti{A}{D}
\end{equation*}
as $D$ ranges over unbiased bits generated from $B$.
It can be shown that $F$ is increasing and
concave (cf.\ \autoref{lem:comparison-calculus}), and
\begin{align*}
  \muti{B}{C\emid U=u}&\le\tsent{C\emid U=u},\\
  \muti{A}{C\emid U=u}
  &\le\tsent{C\emid U=u}
  F\nparen{\frac{\muti{B}{C\emid U=u}}
  {\tsent{C\emid U=u}}}.
\end{align*}
Values of $u$ with zero conditional Tsallis entropy contribute
zero to both informations and may be omitted.
The first inequality, averaged over $U$, places the right-hand
side of \eqref{eq:comparison-match} in $[0,1]$.
Varying the noise in the channel from $B$ to $C'$ therefore
gives a matching bit.
Applying Jensen's inequality to the second inequality, with weights
proportional to $\tsent{C\emid U=u}$, gives
\begin{equation*}
  \muti{A}{C\emid U}
  \le\tsent{C\emid U}
  F\nparen{\frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}}
  =\tsent{C\emid U}\muti{A}{C'},
\end{equation*}
as required.
\end{proof}

\begin{lemma}
\label{lem:comparison-correlation}
Every bit $C'$ satisfying \eqref{eq:comparison-match} also satisfies
\begin{equation*}
  \tsmuti{C'}{B}\ge
  \frac{\tsmuti{C}{B}^{\phantom{2}}}{\tsent{C\emid U}^{2}}.
\end{equation*}
\end{lemma}
\begin{proof}
Let us write $v=\tsent{C\emid U}$ and
$d=\E[(2B-1)(2C-1)]$.
Since $B$ is unbiased, $\tsmuti{C}{B}=d^2$.
As shown in \autoref{app:comparison}, $|d|\le v$ and
\begin{equation*}
  \frac{\muti{B}{C\emid U}}v
  \ge\kldivb{(1+d/v)/2}{1/2}.
\end{equation*}
If $\theta=\E[(2B-1)(2C'-1)]$, then
$\tsmuti{C'}{B}=\theta^2$ and
$\muti{B}{C'}=\kldivb{(1+\theta)/2}{1/2}$.
The matching identity and monotonicity of this divergence in
$|\theta|$ imply $\theta^2\ge d^2/v^2$.
\end{proof}

\subsection{The induction step}
\label{sec:ind-step}
Recalling that $A=X_i$, $B=Y_i$ and $U=X_{-i}$,
in this section we would like to lower bound the gap
$\qent{C} - \muti{C}{X}$ by a factor of the conditional gap
\begin{align*}
\Delta=\qent{C\emid B} - \muti{C}{X\emid B}.
\end{align*}

We obtain two lower bounds on $\qent{C}-\muti{C}{X}$.
The first involves $\Delta$ and benefits from $C$ depending on
$U$, through a positive term in $\tsmuti{C}{U}$.
The second has a negative term in $\tsmuti{C}{U}$ and
benefits from weaker dependence.
We add $\lambda$ times the
second bound to the first, choosing $\lambda$ to cancel these
terms, so that the bound works whether or not $C$ depends on $U$.

\input{figures/comparison-bit}

To analyze the remaining terms, we introduce an unbiased
comparison bit $C'$ obtained from $B$ using private randomness.
The comparisons in \autoref{lem:comparison} and
\autoref{lem:comparison-correlation}, together with our choice
of coordinate with $\tsmuti{C}{B}\le1/4$, allow us to charge
these terms to quantities
involving $\muti{A}{C'}$ and $\muti{B}{C'}$.
What remains is an inequality about three unbiased bits 
forming the Markov chain $A-B-C'$, which we analyze in
\autoref{sec:3bit}.

Let us start with a lemma that we use in obtaining the first
lower bound.
\begin{lemma}
\label{lem:conditioning-gain}
If $\tsent{C\emid U}>0$, then
\begin{equation*}
  \muti{B}{C\emid U}-\muti{B}{C}
  \ge
  \frac{1}{2\ln 2}
  \frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
  {\tsent{C}\,\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
Since $B$ and $U$ are independent, the Shannon and Tsallis
chain rules give
\begin{align*}
  \muti{B}{C\emid U}-\muti{B}{C}
  &=\muti{B}{U\emid C},\\
  \tsmuti{B}{C\emid U}-\tsmuti{B}{C}
  &=\tsmuti{B}{U\emid C}.
\end{align*}
Since $B$ is unbiased and independent of $U$, parts
\ref{item:tsallis-shannon}--\ref{item:tsallis-conditioning}
of \autoref{lem:tsallis-properties} give
\begin{align*}
  \muti{B}{U\emid C}
  &\ge\frac{\tsmuti{B}{U\emid C}}{2\ln2}\\
  &\ge\frac{\tsmuti{C}{B}}{2\ln2}
  \sparen{\frac1{\tsent{C\emid U}}-\frac1{\tsent{C}}}\\
  &=\frac{1}{2\ln 2}
    \frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
         {\tsent{C}\,\tsent{C\emid U}},
\end{align*}
as required.
\end{proof}

When $\tsent{C\emid U}>0$, let $C'$ be an unbiased bit obtained
from $B$ alone using private randomness and satisfying
\eqref{eq:comparison-match}.

\begin{lemma}
\label{lem:gap-first}
If $\tsent{C\emid U}>0$, then
\begin{equation}
  \label{eq:gap-first}
  \begin{aligned}
    \qent{C}-\muti{C}{X}\ge \Delta
    &+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}\\
    &+\frac{(1-\qI)\tsmuti{C}{B}\,\tsmuti{C}{U}}
    {2\ln2\,\tsent{C}\,\tsent{C\emid U}}
    -\qJ\tsmuti{C}{B}.
  \end{aligned}
\end{equation}
\end{lemma}
\begin{proof}
The chain rules for Shannon and Tsallis information give
\begin{align*}
  \qent{C}-\muti{C}{X}-\Delta
  =\muti{C}{B\emid X}
  -(1-\qI)\muti{B}{C}-\qJ\tsmuti{C}{B}.
\end{align*}
The chain rule and \autoref{lem:comparison} give
\begin{align*}
  \muti{C}{B\emid X}
  &=\muti{B}{C\emid U}-\muti{A}{C\emid U}\\
  &\ge(1-\qI)\muti{B}{C\emid U}
  +\tsent{C\emid U}
  \sparen{\qI\muti{B}{C'}-\muti{A}{C'}}.
\end{align*}
Substituting this into the preceding identity and applying
\autoref{lem:conditioning-gain} gives \eqref{eq:gap-first}.
\end{proof}

\begin{lemma}
\label{lem:gap-second}
If $\tsent{C\emid U}>0$, then
\begin{align}
  \label{eq:gap-second}
  \qent{C}-\muti{C}{X}
  &\ge\tsent{C\emid U}\muti{A}{B\emid C'}
  -\sparen{\frac{(1-\qI)c(\mu)}{\ln2}+\qJ}\tsmuti{C}{U}.
\end{align}
\end{lemma}
\begin{proof}
Using the comparison in
$\muti{C}{X}=\muti{C}{U}+\muti{A}{C\emid U}$ gives
\begin{align*}
  \qent{C}-\muti{C}{X}
  &\ge\qI\ent{C}-\qJ\tsent{C}
  -\muti{C}{U}-\tsent{C\emid U}\muti{A}{C'}\\
  &=\tsent{C\emid U}\muti{A}{B\emid C'}
  +\qI\sparen{\ent{C\emid U}-\tsent{C\emid U}}\\
  &\quad-(1-\qI)\muti{C}{U}-\qJ\tsmuti{C}{U}.
\end{align*}
Here we used $\sI=\muti{A}{B}$ and
$\muti{A}{B\emid C'}=\muti{A}{B}-\muti{A}{C'}$.
Now $\ent{C\emid U}\ge\tsent{C\emid U}$ and
\autoref{lem:biased-comparison} give \eqref{eq:gap-second}.
\end{proof}

For $0<p<1$, we use the shorthand
\begin{equation*}
  \alpha_p=\frac{\qJ\ln2}{1-\qI}.
\end{equation*}
We need the following fact about the Markov chain $A-B-C'$
of three unbiased bits. We state it here and prove it in
\autoref{sec:3bit}.

\begin{lemma}
\label{lem:three-bit}
Let $C'$ be an unbiased bit obtained from $B$ using private
randomness. If $0<\qI<1$ and $\tsmuti{B}{C'}>0$, then
\begin{align*}
  \frac{\sI-\muti{A}{C'}}
  {2(1+\alpha_p)}
  +
  \frac{2\nparen{\qI\muti{B}{C'}-\muti{A}{C'}}}
  {\sqrt{\tsmuti{B}{C'}}}
  \ge\qJ.
\end{align*}
\end{lemma}

\begin{proof}[Proof of \autoref{lem:ind-step}]
The cases $p\in\{0,1/2,1\}$ are immediate, so we may assume
$0<p<1/2$, complementing $Y$ if necessary.
If $\tsmuti{C}{B}=0$, then $C$ and $B$ are independent.
Consequently $\qent{C}=\qent{C\emid B}$ and
$\muti{C}{X\emid B}\ge\muti{C}{X}$, proving the claim with
$\lambda=0$.

Therefore we may assume $\tsmuti{C}{B}>0$.
Then we have $\tsent{C\emid U}>0$ as otherwise $C$ would be
determined by $U$ and hence independent of $B$. By
\autoref{lem:comparison}, there is an unbiased bit $C'$
obtained from $B$ alone and satisfying
\eqref{eq:comparison-match}.
Let us choose
\begin{equation*}
  \lambda=
  \frac{\tsmuti{C}{B}}
  {2\tsent{C}\,\tsent{C\emid U}
  \nparen{c(\mu)+\alpha_p}}.
\end{equation*}
Note that $\lambda\ge 0$. When we multiply \eqref{eq:gap-second} by
$\lambda$ and add it to \eqref{eq:gap-first},
the terms involving $\tsmuti{C}{U}$ cancel, giving
\begin{equation}
  \label{eq:gap-combined}
  \begin{aligned}
    (1+\lambda)\nparen{\qent{C}-\muti{C}{X}}\ge\Delta
    &+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}\\
    &+\lambda\tsent{C\emid U}\muti{A}{B\emid C'}
    -\qJ\tsmuti{C}{B}.
  \end{aligned}
\end{equation}

Since $B,C'$ are unbiased, $\tsmuti{C'}{B}=\tsmuti{B}{C'}$.
By \autoref{lem:comparison-correlation} and our choice of coordinate,
\begin{equation*}
  0<\frac{\tsmuti{C}{B}}{\tsent{C\emid U}}
  \le\sqrt{\tsmuti{C}{B}\,\tsmuti{B}{C'}}
  \le\frac12\sqrt{\tsmuti{B}{C'}}.
\end{equation*}
Also, $\tsent{C}\,c(\mu)\le1$ by \autoref{lem:biased-comparison}
and $\tsent{C}\le1$, so
\begin{equation*}
  \frac{\lambda\tsent{C\emid U}}{\tsmuti{C}{B}}
  =\frac1{2\tsent{C}
  \nparen{c(\mu)+\alpha_p}}
  \ge\frac1{2(1+\alpha_p)}.
\end{equation*}
The binary divergence series \eqref{eq:binary-divergence-series} gives
$\qI\muti{B}{C'}-\muti{A}{C'}\ge0$, and
$\muti{A}{B\emid C'}=\sI-\muti{A}{C'}\ge0$.
Substituting the two coefficient bounds in \eqref{eq:gap-combined} yields
\begin{align*}
 (1+\lambda)\nparen{\qent{C}-\muti{C}{X}}
 \ge \Delta+\tsmuti{C}{B}\sparen{
  \frac{\sI-\muti{A}{C'}}{2(1+\alpha_p)}
  +\frac{2\nparen{\qI\muti{B}{C'}-\muti{A}{C'}}}{\sqrt{\tsmuti{B}{C'}}}
  -\qJ}.
\end{align*}
The bracket is nonnegative by \autoref{lem:three-bit}. Thus
$\qent{C}-\muti{C}{X}\ge\Delta/(1+\lambda)$, as required.
\end{proof}

\subsection{The three-bit lemma}
\label{sec:3bit}
Let $A-B-C$ be a Markov chain of unbiased bits. Denote the absolute
correlations of $A,B$ and $B,C$ by $\rho$ and $\sigma$, respectively.
Thus $\rho^2=\tsmuti{A}{B}$ and $\sigma^2=\tsmuti{B}{C}$.
\input{figures/three-bit-chain}

\begingroup
\newtheorem*{restatedthreebitlemma}{Lemma~\ref{lem:three-bit}}
\begin{restatedthreebitlemma}[restated]
If $\tsmuti{A}{B}<1$ and $\tsmuti{B}{C}>0$, then
\begin{equation*}
  \frac{\muti{A}{B\emid C}}
  {2\nparen{1+
    \frac{\nparen{\tsmuti{A}{B}-\muti{A}{B}}\ln2}{\tsent{A\emid B}}}}
  +\frac{2\nparen{\tsmuti{A}{B}\muti{B}{C}-\muti{A}{C}}}
  {\sqrt{\tsmuti{B}{C}}}
  \ge\tsmuti{A}{B}-\muti{A}{B}.
\end{equation*}
\end{restatedthreebitlemma}
\endgroup
\begin{proof}
If $\rho=0$, both sides vanish. Assume $0<\rho<1$ and $0<\sigma\le1$.
Write $f(t)=\kldivb{(1+t)/2}{1/2}$. The Markov property gives
\begin{align*}
 \sI=\muti{A}{B}=f(\rho),\qquad \qI=\rho^2,\qquad
 \muti{B}{C}=f(\sigma),\qquad \muti{A}{C}=f(\rho\sigma).
\end{align*}
Since $\muti{A}{B\emid C}=\sI-\muti{A}{C}$ and $\tsent{A\emid B}=1-\qI$,
it suffices to show
\begin{equation}
 \label{eq:three-bit-scalar-target}
 \frac{\sI-f(\rho\sigma)}{2(1+\alpha_p)}
 +2\frac{\qI f(\sigma)-f(\rho\sigma)}{\sigma}>\qJ .
\end{equation}
The series \eqref{eq:binary-divergence-series} gives
$f(t)\ln2=\sum_{k\ge1}a_k t^{2k}$, where $a_k=1/[2k(2k-1)]$.
For $j\ge1$ write
\begin{equation*}
 s_j(\sigma)=\sum_{k>j}a_k\sigma^{2k},\qquad b_j=s_j(1),\qquad
 \alpha_p=\frac{\qJ\ln2}{1-\qI}=\sum_{j\ge1}b_j\qI^j .
\end{equation*}
The last identity follows by expanding
$\qI-\qI^k=(1-\qI)\sum_{j=1}^{k-1}\qI^j$ and using $\sum_{k\ge1}a_k=\ln2$.
In particular $\qJ>0$.

We apply Jensen's inequality to both terms of \eqref{eq:three-bit-scalar-target}
at a common exponent. Let $k_*$ and $j_*$ be the means of the distributions
on positive integers with weights proportional to $k a_k\qI^k$ and $b_j\qI^j$,
respectively, and set
\begin{equation*}
 h=\frac{2b_1\sum_{k\ge1} k a_k\qI^k}{(1+\alpha_p)\qJ\ln2}.
\end{equation*}
We prove below that $k_*\le j_*\le h$ and that
\begin{equation}
 \label{eq:scalar-rescaled-order}
 \frac{s_j(\sigma)}{\sigma b_j}\ge
 \frac{s_1(\sigma^j)}{\sigma^j b_1}.
\end{equation}
Integrating the series for $f$ and applying Jensen's inequality
to $k\mapsto e^{-kt}$ gives, since $k_*\le h$,
\begin{align*}
 \bigl[\sI-f(\rho\sigma)\bigr]\ln2
 &=\int_0^{-2\ln\sigma}\sum_{k\ge1} k a_k\qI^k e^{-kt}\,\mathrm dt
 \ge \frac{1-\sigma^{2h}}{h}\sum_{k\ge1} k a_k\qI^k .
\end{align*}
For the second term, $j\mapsto s_1(\sigma^j)/\sigma^j$ is a decreasing
convex sum of exponentials. Using \eqref{eq:scalar-rescaled-order},
Jensen's inequality, and $j_*\le h$ gives
\begin{equation*}
 \frac{\qI f(\sigma)-f(\rho\sigma)}{\sigma\qJ}
 =\frac{\sum_{j\ge1}\qI^j s_j(\sigma)}{\sigma\alpha_p}
 \ge \frac{s_1(\sigma^{j_*})}{\sigma^{j_*} b_1}
 \ge \frac{s_1(\sigma^h)}{\sigma^h b_1}.
\end{equation*}
Substituting in \eqref{eq:three-bit-scalar-target} and writing $t=\sigma^h$,
we need only prove
\begin{equation*}
 \frac{1-t^2}{2}+4\frac{s_1(t)}{t}>2b_1.
\end{equation*}
Keep the cubic and quintic terms in $4s_1(t)/t$. With $y=t^2$, their
sum is the convex function $g(y)=y^{3/2}/3+2y^{5/2}/15$.
At $y=a^2$ for $a=11/15$, the tangent has slope $a/2+a^3/3<1/2$.
Adding $(1-y)/2$ to this tangent and taking its minimum at $y=1$ gives
\begin{equation*}
 (1-y)/2+g(y)\ge a/2+a^3/6-a^5/5>7/18>2\ln2-1=2b_1 .
\end{equation*}

\emph{Rescaling the tails.}
Integrating the alternating harmonic tail gives
$b_j=\int_0^1 u^{2j}/(1+u)\,\mathrm du$, and summing a geometric series gives
$s_j(\sigma)=\int_0^\sigma(\sigma-u)u^{2j}/(1-u^2)\,\mathrm du$.
Write $\sigma=e^{-a/j}$. Substituting $u=e^{-z/j}$ in the first integral
and $u=e^{-(z+a)/j}$ in the second gives
\begin{equation*}
 \frac{s_j(e^{-a/j})}{e^{-a/j}b_j}
 =e^{-2a}\E_{\pi_j}\frac{\sinh(z/j)}{\sinh((z+a)/j)},\qquad
 \pi_j(\mathrm dz)\propto\frac{e^{-2z}}{1+e^{z/j}}\,\mathrm dz,\quad z>0.
\end{equation*}
The quotient increases in $z$ since $\coth$ decreases, and increases in
$j$ since $u\coth u$ increases on $(0,\infty)$. For $\ell>j$, the density
ratio $\pi_\ell/\pi_j$ also increases in $z$, since its logarithmic derivative is
\begin{equation*}
 \frac{1/j}{1+e^{-z/j}}-\frac{1/\ell}{1+e^{-z/\ell}}>0 .
\end{equation*}
Thus increasing $j$ increases the expectation, by the nonnegative covariance
of two increasing functions of $z$. Comparing with $j=1$ at the same $a$
proves \eqref{eq:scalar-rescaled-order}.

\emph{Comparing the means.}
We have $k_*\le j_*$ because the ratio of the two weights is proportional to
\begin{equation*}
 \frac{b_j}{j a_j}
 =2\int_0^1 (2j-1)t^{2j-2}\frac{t^2}{1+t}\,\mathrm dt .
\end{equation*}
This is twice the expectation of an increasing function under the
distribution with cumulative distribution function $t^{2j-1}$ on $[0,1]$.
The expectation increases with $j$.

For $j_*\le h$, substitute $j_*=\sum_{j\ge1}j b_j\qI^j/\alpha_p$ and
$\qJ\ln2=(1-\qI)\alpha_p$. The required inequality is
\begin{equation*}
 (1+\alpha_p)\sum_{j\ge1}j b_j\qI^j
 \le\frac{2b_1}{1-\qI}\sum_{k\ge1} k a_k\qI^k .
\end{equation*}
Put $b_0=1$ and $e_n=\sum_{k=1}^n k a_k$. Comparing coefficients of
$\qI^n$, it suffices that
\begin{equation*}
 \sum_{j=1}^n j b_j b_{n-j}\le 2b_1e_n .
\end{equation*}
There is equality at $n=1$. Telescoping $a_k\le1/[4k(k-1)]$ for $k\ge2$
gives $j b_j\le1/4$. Summing the tails gives
$\sum_{j=1}^{n-1}b_j=e_n+n b_n-\ln2$. Consequently
\begin{equation*}
 \sum_{j=1}^n j b_j b_{n-j}
 \le \tfrac14\left(1+\sum_{j=1}^{n-1}b_j\right)
 \le \tfrac14(e_n+5/4-\ln2).
\end{equation*}
For $n\ge9$ this is at most $2b_1 e_n$, since $e_n\ge e_9>1.03$ and
$\ln2>0.6931$.

For $2\le n\le8$, use $0.6931<\ln2<0.6932$, which follows by summing
$2\sum_{j\ge0}3^{-2j-1}/(2j+1)$ and bounding the tail geometrically.
In $2b_1e_n-\sum_{j=1}^n j b_j b_{n-j}$ use the lower endpoint for
$b_1=\ln2-1/2$ in the first term and the upper endpoint for each
$b_j=\ln2-\sum_{k=1}^j a_k$, $j\ge1$, in the sum. The resulting rational
lower bounds exceed, in order of increasing $n$,
\begin{equation*}
 10^{-5}(40,280,565,855,1135,1395,1640).
\end{equation*}
This gives $j_*\le h$ and completes the proof.
\end{proof}
