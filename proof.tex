\section{The proof}
\label{sec:proof}
Our proof proceeds by induction on $n$, the number of
coordinates of $X,Y$. Given the random bit $C$ and the
bit string $Y\!$, we adaptively pick a coordinate $i\in[n]$ and
establish the bound
\begin{align}
  \tag{($\star$)}\label{eq:star}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid Y_i}-\muti{C}{X\emid Y_i}}
\end{align}
for some $\lambda\ge 0$.
When $Y_i$ is fixed, $C$ depends on $n-1$ coordinates and
therefore the induction hypothesis applies for $Y_i=0$ and
$Y_i=1$ separately, showing that
$\qent{C\emid Y_i}\ge \muti{C}{X\emid Y_i}$.
Plugging into \eqref{eq:star}, we get
$\qent{C}\ge\muti{C}{X}$, which implies \eqref{eq:ck}
as we show next.
\begin{lemma}
\label{lem:ck+to_ck}
For all $p\in[0,1]$, we have
$\qent{C}\le \muti{Y_1}{X}$.
\end{lemma}
\begin{proof}
From \hyperref[item:tsallis-noise]{Lemma~\ref*{lem:tsallis-properties}(\ref*{item:tsallis-noise})}
and \hyperref[item:tsallis-bounds]{(\ref*{item:tsallis-bounds})},
we have $\qJ\le\sI$ and
$\ent{C}-\tsent{C}\le1-\ent{C}$. Let us write
\begin{align*}
  \qent{C} - \sI\ent{C}
  &= \qJ\ent{C} - \qJ\tsent{C}\\
  &=   \qJ\nparen{\ent{C}-\tsent{C}}\\
  &\le \sI \nparen{1-\ent{C}}.
\end{align*}
Therefore
$\Q\le \sI\nparen{1-\ent{C}} + \sI\,\ent{C} = \muti{Y_1}{X}$
as desired.
\end{proof}
The property we want from the coordinate $i$ is fairly mild:
$\tsmuti{C}{Y_i}\le 1/4$. The next lemma shows that such a
coordinate always exists.

\begin{lemma}
\label{lem:good-coordinate}
When $n\ge2$, there exists an $i\in[n]$ such that
$\tsmuti{C}{Y_i}\le 1/4$.
\end{lemma}
\begin{proof}
Let $i,j\in[n]$ be distinct. We may relabel $Y_i,Y_j$ so that
$\E[C(2Y_i-1)]$ and $\E[C(2Y_j-1)]$ are nonnegative. Since
$Y_i,Y_j$ are independent uniform bits,
\begin{equation*}
  \sqrt{\tsmuti{C}{Y_i}}+\sqrt{\tsmuti{C}{Y_j}}
  =2\E\sparen{C(2Y_i-1)+C(2Y_j-1)}
  \le4\Pr[Y_i=Y_j=1]=1,
\end{equation*}
using $0\le C\le1$. At least one of the two square roots
is at most $1/2$.
\end{proof}
Let us fix such an $i$ satisfying $\tsmuti{C}{Y_i}\le1/4$
and use the shorthands $A=X_i$, $B=Y_i$, and $U= X_{-i}$
in the rest of the paper.
The following lemma is the crux of our argument,
which establishes \eqref{eq:star} giving the induction step.
\begin{lemma}
\label{lem:ind-step}
Assuming $\tsmuti{C}{B}\le1/4$, we have
\begin{align*}
  \qent{C} - \muti{C}{X} \ge \frac{1}{1+ \lambda}
  \nparen{\qent{C\emid B}-\muti{C}{X\emid B}}
\end{align*}
for some $\lambda\ge 0$.
\end{lemma}
\autoref{sec:ind-step} is devoted to proving \autoref{lem:ind-step};
here we show that it implies the main theorem.
\begingroup
\newtheorem*{restatedtheorem}{Theorem~\ref{thm:ck+}}
\begin{restatedtheorem}[restated]
It holds that $\muti{C}{X}\le \qent{C}\le\muti{Y_1}{X}$.
\end{restatedtheorem}
\endgroup
\begin{proof}
The second inequality is \autoref{lem:ck+to_ck}.
For the first, the cases $p\in\{0,1/2,1\}$ and constant $C$
are immediate. Complementing $Y$ if necessary lets us assume
$0<p<1/2$. We proceed by induction on $n$.
When $n=1$, the bit $C$ is a constant, $Y_1$, or its negation,
and the bound holds with equality.
For $n\ge2$, by \autoref{lem:good-coordinate} there exists a coordinate
$i$ with $\tsmuti{C}{Y_i}\le1/4$.
Conditioning on $B=Y_i$ reduces the problem to $n-1$ coordinates,
so the induction hypothesis applies to the $B=0$ and $B=1$ cases
individually, giving
\begin{equation*}
  \muti{C}{X\emid B}=\muti{C}{U\emid B}
  \le\qent{C\emid B}.
\end{equation*}
Applying \autoref{lem:ind-step}, we conclude
$\qent{C}-\muti{C}{X}\ge0$.
\end{proof}

\subsection{A comparison lemma}
\label{sec:comparison}

Let $U$ be a random variable and $A,B$ be unbiased bits jointly
independent of $U$. Let $C$ be a bit obtained from $(B,U)$ using
private randomness, with $\tsent{C\emid U}>0$.
We compare $C$ with an unbiased bit $C'$ obtained from $B$ alone
using private randomness, subject to the matching condition
\begin{equation}
  \tag{(B)}
  \label{eq:comparison-match}
  \muti{B}{C'}=
  \frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}.
\end{equation}

\begin{lemma}
\label{lem:comparison}
A bit $C'$ satisfying \eqref{eq:comparison-match} exists.
Every such bit satisfies
\begin{equation*}
  \muti{A}{C'}\ge
  \frac{\muti{A}{C\emid U}}{\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
For the fixed channel between $A$ and $B$, let $F$ be the
function determined by
\begin{equation*}
F\bigl(\muti{B}{D}\bigr)=\muti{A}{D}
\end{equation*}
as $D$ ranges over unbiased bits generated from $B$.
It can be shown that $F$ is increasing and
concave (cf.\ \autoref{lem:comparison-calculus}), and
\begin{align*}
  \muti{B}{C\emid U=u}&\le\tsent{C\emid U=u},\\
  \muti{A}{C\emid U=u}
  &\le\tsent{C\emid U=u}
  F\left(\frac{\muti{B}{C\emid U=u}}
  {\tsent{C\emid U=u}}\right).
\end{align*}
Values of $u$ with zero conditional Tsallis entropy contribute
zero to both informations and may be omitted.
The first inequality, averaged over $U$, places the right-hand
side of \eqref{eq:comparison-match} in $[0,1]$.
Varying the noise in the channel from $B$ to $C'$ therefore
gives a matching bit.
Applying Jensen's inequality to the second inequality, with weights
proportional to $\tsent{C\emid U=u}$, gives
\begin{equation*}
  \muti{A}{C\emid U}
  \le\tsent{C\emid U}
  F\left(\frac{\muti{B}{C\emid U}}{\tsent{C\emid U}}\right)
  =\tsent{C\emid U}\muti{A}{C'},
\end{equation*}
as required.
\end{proof}

\begin{lemma}
\label{lem:comparison-correlation}
Every bit $C'$ satisfying \eqref{eq:comparison-match} also satisfies
\begin{equation*}
  \tsmuti{C'}{B}\ge
  \frac{\tsmuti{C}{B}^{\phantom{2}}}{\tsent{C\emid U}^{2}}.
\end{equation*}
\end{lemma}
\begin{proof}
Let us write $v=\tsent{C\emid U}$ and
$d=\E[(2B-1)(2C-1)]$.
Since $B$ is unbiased, $\tsmuti{C}{B}=d^2$.
As shown in \autoref{app:comparison}, $|d|\le v$ and
\begin{equation*}
  \frac{\muti{B}{C\emid U}}v
  \ge\kldivb{(1+d/v)/2}{1/2}.
\end{equation*}
If $\theta=\E[(2B-1)(2C'-1)]$, then
$\tsmuti{C'}{B}=\theta^2$ and
$\muti{B}{C'}=\kldivb{(1+\theta)/2}{1/2}$.
The matching identity and monotonicity of this divergence in
$|\theta|$ imply $\theta^2\ge d^2/v^2$.
\end{proof}

\subsection{The induction step}
\label{sec:ind-step}
Recalling that $A=X_i$, $B=Y_i$ and $U=X_{-i}$,
in this section we would like to lower bound the gap
$\qent{C} - \muti{C}{X}$ by a factor of the conditional gap
\begin{align*}
\Delta=\qent{C\emid B} - \muti{C}{X\emid B}.
\end{align*}

We obtain two lower bounds on $\qent{C}-\muti{C}{X}$.
The first involves $\Delta$ and benefits from $C$ depending on
$U$, through a positive term in $\tsmuti{C}{U}$.
The second has a negative term in $\tsmuti{C}{U}$ and
benefits from weaker dependence.
We add $\lambda$ times the
second bound to the first, choosing $\lambda$ to cancel these
terms, so that the bound works whether or not $C$ depends on $U$.

To analyze the remaining terms, we introduce an unbiased
comparison bit $C'$ obtained from $B$ using private randomness.
The comparisons in \autoref{lem:comparison} and
\autoref{lem:comparison-correlation}, together with our choice
of coordinate with $\tsmuti{C}{B}\le1/4$, allow us to charge
these terms to quantities
involving $\muti{A}{C'}$ and $\muti{B}{C'}$.
What remains is an inequality about the three unbiased bits in the
Markov chain $A-B-C'$, which we analyze in \autoref{sec:3bit}.

Let us start with a lemma that we use in obtaining the first
lower bound.
\begin{lemma}
\label{lem:conditioning-gain}
If $\tsent{C\emid U}>0$, then
\begin{equation*}
  \muti{B}{C\emid U}-\muti{B}{C}
  \ge
  \frac{1}{2\ln 2}
  \frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
  {\tsent{C}\,\tsent{C\emid U}}.
\end{equation*}
\end{lemma}
\begin{proof}
Since $B$ and $U$ are independent, the Shannon and Tsallis
chain rules give
\begin{align*}
  \muti{B}{C\emid U}-\muti{B}{C}
  &=\muti{B}{U\emid C},\\
  \tsmuti{B}{C\emid U}-\tsmuti{B}{C}
  &=\tsmuti{B}{U\emid C}.
\end{align*}
Since $B$ is unbiased and independent of $U$, parts
\ref{item:tsallis-shannon}--\ref{item:tsallis-conditioning}
of \autoref{lem:tsallis-properties} give
\begin{align*}
  \muti{B}{U\emid C}
  &\ge\frac{\tsmuti{B}{U\emid C}}{2\ln2}\\
  &\ge\frac{\tsmuti{C}{B}}{2\ln2}
  \left(\frac1{\tsent{C\emid U}}-\frac1{\tsent{C}}\right)\\
  &=\frac{\tsmuti{C}{B}\,\tsmuti{C}{U}}
  {2\ln2\,\tsent{C}\,\tsent{C\emid U}},
\end{align*}
as required.
\end{proof}

When $\tsent{C\emid U}>0$, let $C'$ be an unbiased bit obtained
from $B$ alone using private randomness and satisfying
\eqref{eq:comparison-match}.

\begin{lemma}
\label{lem:gap-first}
If $\tsent{C\emid U}>0$, then
\begin{equation}
  \label{eq:gap-first}
  \begin{aligned}
    \qent{C}-\muti{C}{X}\ge \Delta
    &+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}\\
    &+\frac{(1-\qI)\tsmuti{C}{B}\,\tsmuti{C}{U}}
    {2\ln2\,\tsent{C}\,\tsent{C\emid U}}
    -\qJ\tsmuti{C}{B}.
  \end{aligned}
\end{equation}
\end{lemma}
\begin{proof}
The chain rules for Shannon and Tsallis information give
\begin{align*}
  \qent{C}-\muti{C}{X}-\Delta
  =\muti{C}{B\emid X}
  -(1-\qI)\muti{B}{C}-\qJ\tsmuti{C}{B}.
\end{align*}
The chain rule and \autoref{lem:comparison} give
\begin{align*}
  \muti{C}{B\emid X}
  &=\muti{B}{C\emid U}-\muti{A}{C\emid U}\\
  &\ge(1-\qI)\muti{B}{C\emid U}
  +\tsent{C\emid U}
  \bigl[\qI\muti{B}{C'}-\muti{A}{C'}\bigr].
\end{align*}
Substituting this into the preceding identity and applying
\autoref{lem:conditioning-gain} gives \eqref{eq:gap-first}.
\end{proof}

\begin{lemma}
\label{lem:gap-second}
If $\tsent{C\emid U}>0$, then
\begin{align}
  \label{eq:gap-second}
  \qent{C}-\muti{C}{X}
  &\ge\tsent{C\emid U}\muti{A}{B\emid C'}
  -\left[\frac{(1-\qI)c(\mu)}{\ln2}+\qJ\right]\tsmuti{C}{U}.
\end{align}
\end{lemma}
\begin{proof}
Using the comparison in
$\muti{C}{X}=\muti{C}{U}+\muti{A}{C\emid U}$ gives
\begin{align*}
  \qent{C}-\muti{C}{X}
  &\ge\qI\ent{C}-\qJ\tsent{C}
  -\muti{C}{U}-\tsent{C\emid U}\muti{A}{C'}\\
  &=\tsent{C\emid U}\muti{A}{B\emid C'}
  +\qI\bigl[\ent{C\emid U}-\tsent{C\emid U}\bigr]\\
  &\quad-(1-\qI)\muti{C}{U}-\qJ\tsmuti{C}{U}.
\end{align*}
Here we used $\sI=\muti{A}{B}$ and
$\muti{A}{B\emid C'}=\muti{A}{B}-\muti{A}{C'}$.
Now $\ent{C\emid U}\ge\tsent{C\emid U}$ and
\autoref{lem:biased-comparison} give \eqref{eq:gap-second}.
\end{proof}

For $0<p<1$, we use the shorthand
\begin{equation*}
  \alpha_p=\frac{\qJ\ln2}{1-\qI}.
\end{equation*}
The following fact about the unbiased chain $A-B-C'$ is the
remaining ingredient. We state it here and prove it in
\autoref{sec:3bit}.

\begin{lemma}
\label{lem:three-bit}
Let $C'$ be an unbiased bit obtained from $B$ using private
randomness. If $0<\qI<1$ and $\tsmuti{B}{C'}>0$, then
\begin{align*}
  \frac{\sI-\muti{A}{C'}}
  {2(1+\alpha_p)}
  +
  \frac{2\nparen{\qI\muti{B}{C'}-\muti{A}{C'}}}
  {\sqrt{\tsmuti{B}{C'}}}
  \ge\qJ.
\end{align*}
\end{lemma}

\begin{proof}[Proof of \autoref{lem:ind-step}]
The cases $p\in\{0,1/2,1\}$ are immediate, so we may assume
$0<p<1/2$, complementing $Y$ if necessary.
If $\tsmuti{C}{B}=0$, then $C$ and $B$ are independent.
Consequently $\qent{C}=\qent{C\emid B}$ and
$\muti{C}{X\emid B}\ge\muti{C}{X}$, proving the claim with
$\lambda=0$.

We may therefore assume $\tsmuti{C}{B}>0$.
Then $\tsent{C\emid U}>0$:
otherwise $C$ would be determined by $U$ and hence independent of
$B$. By \autoref{lem:comparison}, there is an unbiased bit $C'$
obtained from $B$ alone and satisfying \eqref{eq:comparison-match}.
Let us write
\begin{equation*}
  \lambda=
  \frac{\tsmuti{C}{B}}
  {2\tsent{C}\,\tsent{C\emid U}
  \left(c(\mu)+\alpha_p\right)}.
\end{equation*}
All factors are positive. When we multiply \eqref{eq:gap-second} by
$\lambda$ and add it to \eqref{eq:gap-first},
the terms involving $\tsmuti{C}{U}$ cancel, giving
\begin{equation}
  \label{eq:gap-combined}
  \begin{aligned}
    &(1+\lambda)\bigl[\qent{C}-\muti{C}{X}\bigr]\\
    &\quad\ge\qent{C\emid B}-\muti{C}{X\emid B}
    -\qJ\tsmuti{C}{B}\\
    &\qquad+\tsent{C\emid U}
    \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}
    +\lambda\tsent{C\emid U}\muti{A}{B\emid C'}.
  \end{aligned}
\end{equation}

By \autoref{lem:comparison-correlation} and the choice of coordinate,
\begin{equation*}
  \frac{\tsmuti{C}{B}}{\tsent{C\emid U}}
  \le\sqrt{\tsmuti{C}{B}\,\tsmuti{C'}{B}}
  \le\frac12\sqrt{\tsmuti{C'}{B}}.
\end{equation*}
Also, \autoref{lem:biased-comparison} and $0<\tsent{C}\le1$ give
\begin{equation*}
  \tsent{C}\left(c(\mu)+\alpha_p\right)
  \le1+\tsent{C}\alpha_p
  \le1+\alpha_p.
\end{equation*}
Consequently,
\begin{equation*}
  \frac{\lambda\tsent{C\emid U}}{\tsmuti{C}{B}}
  =\frac1{2\tsent{C}
  \left(c(\mu)+\alpha_p\right)}
  \ge\frac1{2(1+\alpha_p)}.
\end{equation*}
The two information quantities multiplied by these coefficients
are nonnegative. For the first, the binary divergence series gives
$\muti{A}{C'}\le\tsmuti{A}{B}\muti{B}{C'}$; the second
is a conditional mutual information.
Applying \autoref{lem:three-bit}, we therefore obtain
\begin{align*}
  &\frac{\tsent{C\emid U}}{\tsmuti{C}{B}}
  \nparen{\qI\muti{B}{C'}-\muti{A}{C'}}
  +\frac{\lambda\tsent{C\emid U}}{\tsmuti{C}{B}}
  \muti{A}{B\emid C'}\ge\qJ.
\end{align*}
Thus the last three terms in \eqref{eq:gap-combined} have
nonnegative sum. Dividing by $1+\lambda$ proves the lemma.
\end{proof}
